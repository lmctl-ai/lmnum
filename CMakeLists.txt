cmake_minimum_required(VERSION 3.24)
project(lmnum VERSION 0.1.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(LMNUM_FULL_STATIC "Link the complete executable statically" ON)

foreach(dep eigen sundials)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${dep}")
    message(FATAL_ERROR "third_party/${dep} missing: run scripts/sync-deps.sh")
  endif()
endforeach()

# Embed schema and notices.
function(embed_file var file ns name)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  file(READ "${file}" hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
  set(${var} "inline constexpr unsigned char ${name}[] = {${hex}0}; inline constexpr std::size_t ${name}_size = sizeof(${name}) - 1;\n" PARENT_SCOPE)
endfunction()
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}")
embed_file(SCHEMA_SRC "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmnum.dsl.json" lmnum embedded_schema)
file(WRITE "${GEN}/schema_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmnum { ${SCHEMA_SRC} }\n")
embed_file(NOTICES_SRC "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES" lmnum embedded_notices)
file(WRITE "${GEN}/notices_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmnum { ${NOTICES_SRC} }\n")

# SUNDIALS: only CVODE and KINSOL with serial vectors and dense linear algebra, static, no extras.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
foreach(opt BUILD_ARKODE BUILD_CVODES BUILD_IDA BUILD_IDAS EXAMPLES_ENABLE_C EXAMPLES_ENABLE_CXX EXAMPLES_INSTALL
        ENABLE_MPI ENABLE_OPENMP ENABLE_LAPACK ENABLE_KLU ENABLE_PTHREAD SUNDIALS_ENABLE_ERROR_CHECKS)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_CVODE ON CACHE BOOL "" FORCE)
set(BUILD_KINSOL ON CACHE BOOL "" FORCE)
set(SUNDIALS_INDEX_SIZE 64 CACHE STRING "" FORCE)
set(EXAMPLES_ENABLE_C OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/sundials EXCLUDE_FROM_ALL)

add_executable(lmnum cpp/main.cpp cpp/numeric.cpp cpp/memory_budget.cpp)
target_compile_definitions(lmnum PRIVATE LMNUM_VERSION="${PROJECT_VERSION}" EIGEN_MPL2_ONLY)
target_include_directories(lmnum PRIVATE cpp "${GEN}" third_party/eigen)
target_link_libraries(lmnum PRIVATE sundials_cvode_static sundials_kinsol_static sundials_nvecserial_static
                      sundials_sunmatrixdense_static sundials_sunlinsoldense_static sundials_core_static)
target_compile_options(lmnum PRIVATE -Wall -Wextra)
if((CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang") AND NOT APPLE)
  target_link_options(lmnum PRIVATE -static-libstdc++ -static-libgcc)
  if(LMNUM_FULL_STATIC)
    target_link_options(lmnum PRIVATE -static)
  endif()
  if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_link_options(lmnum PRIVATE -s)
  endif()
endif()
